import java.util.Scanner;
 
public class Main {
    public static void main(String[] args) {
 
        Scanner teclado = new Scanner(System.in);
 
        int N, M;
 
        System.out.print("Ingrese la cantidad de articulos: ");
        N = teclado.nextInt();
 
        System.out.print("Ingrese la cantidad de sucursales: ");
        M = teclado.nextInt();
 
        double[][] precios = new double[N][M];
        int[][] cantidades = new int[N][M];
 
        // Ingresar los precios de cada articulo por sucursal
        System.out.println("\n=== INGRESO DE PRECIOS ===");
 
        for (int i = 0; i < N; i++) {
            for (int j = 0; j < M; j++) {
                System.out.print("Precio del articulo " + (i + 1)
                        + " en la sucursal " + (j + 1) + ": ");
                precios[i][j] = teclado.nextDouble();
            }
        }
 
        // Ingresar las cantidades vendidas
        System.out.println("\n=== INGRESO DE CANTIDADES VENDIDAS ===");
 
        for (int i = 0; i < N; i++) {
            for (int j = 0; j < M; j++) {
                System.out.print("Cantidad vendida del articulo "
                        + (i + 1) + " en la sucursal " + (j + 1) + ": ");
                cantidades[i][j] = teclado.nextInt();
            }
        }
 
        // Calcular el total de ventas por articulo
        System.out.println("\n=== TOTAL DE VENTAS POR ARTICULO ===");
 
        for (int i = 0; i < N; i++) {
            double totalArticulo = 0;
 
            for (int j = 0; j < M; j++) {
                totalArticulo += precios[i][j] * cantidades[i][j];
            }
 
            System.out.printf("Articulo %d: $%.2f%n",
                    (i + 1), totalArticulo);
        }
 
        // Calcular el total de ventas por sucursal
        System.out.println("\n=== TOTAL DE VENTAS POR SUCURSAL ===");
 
        for (int j = 0; j < M; j++) {
            double totalSucursal = 0;
 
            for (int i = 0; i < N; i++) {
                totalSucursal += precios[i][j] * cantidades[i][j];
            }
 
            System.out.printf("Sucursal %d: $%.2f%n",
                    (j + 1), totalSucursal);
        }
 
        teclado.close();
    }
}
